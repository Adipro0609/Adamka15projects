const express = require("express"); const session = require("express-session"); const bodyParser = require("body-parser"); const app = express(); // Middleware app.use(bodyParser.urlencoded({ extended: true })); app.use(session({ secret: "secret-key", resave: false, saveUninitialized: true })); // Fake user (for demo) const USER = { username: "admin", password: "1234" }; // Routes app.get("/", (req, res) => { res.sendFile(__dirname + "/login.html"); }); app.post("/login", (req, res) => { const { username, password } = req.body; if (username === USER.username && password === USER.password) { req.session.loggedIn = true; res.redirect("/home"); } else { res.send("Invalid login. Try again"); } }); app.get("/home", (req, res) => { if (req.session.loggedIn) { res.sendFile(__dirname + "/home.html"); } else { res.redirect("/"); } }); app.get("/logout", (req, res) => { req.session.destroy(); res.redirect("/"); }); // Start server app.listen(3000, () => { console.log("Server running on http://localhost:3000"); }); Login
